<script>

    document.getElementById('robertForm').addEventListener('submit', function(e) {

        e.preventDefault(); // 防止畫面跳轉

       

        const form = this;

        const btn = document.querySelector('.submit-btn');

        const originalBtnText = btn.textContent;

       

        // 變更按鈕狀態為發射中

        btn.textContent = '🚀 補給艙發射中...';

        btn.disabled = true;

        btn.style.backgroundColor = '#4A5568';

 

        const formData = new FormData(form);

 

        // 將資料發送到您的 Apps Script

        fetch('https://script.google.com/macros/s/AKfycbz0KYgW_e3hMgoyIkI1avK4X7l5TB0Phzac2WQAluR_k1clMVeKUpaby-7e0wiOy2DsHg/exec', {

            method: 'POST',

            body: formData

        })

        .then(response => response.json())

        .then(data => {

            if(data.result === 'success') {

                alert('✨ 登機成功！Robert 已經收到您的補給需求。');

                form.reset(); // 清空表單

            } else {

                alert('⚠️ 發生錯誤，請稍後再試一次。');

            }

        })

        .catch(error => {

            console.error('Error:', error);

            alert('連線失敗，請檢查網路連線。');

        })

        .finally(() => {

            // 恢復按鈕狀態

            btn.textContent = originalBtnText;

            btn.disabled = false;

            btn.style.backgroundColor = '';

        });

    });

</script>
